import yfinance as yf
import pandas as pd
from concurrent.futures import ThreadPoolExecutor, as_completed
import logging

logger = logging.getLogger(__name__)

def batch_download(symbols: list, period: str = "1y") -> dict:
    """
    Download OHLCV for multiple symbols at once using yf.download batch API.
    Returns dict: {symbol: DataFrame}
    """
    if not symbols:
        return {}
    try:
        raw = yf.download(
            tickers=symbols,
            period=period,
            interval="1d",
            group_by="ticker",
            auto_adjust=True,
            threads=True,
            progress=False,
        )
        result = {}
        for sym in symbols:
            try:
                df = raw[sym].dropna() if len(symbols) > 1 else raw.dropna()
                if len(df) >= 200:
                    result[sym] = df
            except Exception:
                pass
        return result
    except Exception as e:
        logger.warning("Batch download failed: %s", e)
        return {}

def pre_filter(stocks: list) -> list:
    """
    Quick pre-filter: download 1 month of data for all stocks,
    keep only those with price > 20, avg_volume > 200000.
    Run in batches of 100.
    """
    symbols = [s["symbol"] for s in stocks]
    passed = set()
    
    for i in range(0, len(symbols), 100):
        batch = symbols[i:i+100]
        try:
            raw = yf.download(
                tickers=batch,
                period="1mo",
                interval="1d",
                group_by="ticker",
                auto_adjust=True,
                threads=True,
                progress=False,
            )
            for sym in batch:
                try:
                    df = raw[sym].dropna() if len(batch) > 1 else raw.dropna()
                    if df.empty:
                        continue
                    price = float(df["Close"].iloc[-1])
                    avg_vol = float(df["Volume"].mean())
                    if price > 20 and avg_vol > 200000:
                        passed.add(sym)
                except Exception:
                    pass
        except Exception as e:
            logger.warning("Pre-filter batch %d failed: %s", i, e)
    
    return [s for s in stocks if s["symbol"] in passed]
